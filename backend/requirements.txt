#   1. After adding/changing a dependency in pyproject.toml

#   pip install pip-tools
#   pip-compile pyproject.toml --output-file requirements.txt
#   # or with uv (faster):
#   uv pip compile pyproject.toml -o requirements.txt
#
# To install for development:
#   pip install -r requirements.txt
#   pip install -e ".[eval]"       # adds pytest, pytest-asyncio
#
# Docker uses this file: COPY requirements.txt . && pip install -r requirements.txt
